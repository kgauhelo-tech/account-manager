START
    //Declarations
    DECLARE CreateAccount
    DECLARE RequestAmount
    DECLARE DepositAmount
    DECLARE TransferAmount
    DECLARE TargetAccount
    DECLARE AccountBalance
    DECLARE AverageTransactionAmount
    DECLARE LastRequestTIme

    // Input and prompting the user
    CLASS BankAccount:
        private Username
        private AccountNumber
        private AccountBalance
        private LastRequestTIme

        Constructor(Username):
            Username property is assigned the Username parameter value
            AccountNumber is automatically generated
            AccountBalance is 0 by default

        FUNCTION Transfer(TransferAmount, TargetAccount):
            If AccountBalance >= TransferAmount && TransferAmount < AverageTransactionAmount * 5:
                Transfer the amount
                Update the balance

            ELSE:
                Reject transaction
            ENDIF
        END FUNCTION

        FUNCTION Withdraw(RequestAmount):
            IF RequestAmount <= AccountBalance && LastRequestTIme + 10 seconds > now() is true:
                Allow withdrawal
                Update the AccountBalance
                Update LastRequestTIme
                Show success message
            ELSE
                Reject withdrawal
                Display reason for rejection
            ENDIF
        END FUNCTION

        FUNCTION deposit(DepositAmount):
            Add DepositAmount to AccountBalance
            Give success message
        END FUNCTION

        FUNCTION BalanceCheck():
            Return AccountBalance
        END FUNCTION
    END CLASS


    // Creating an account
    Create a new instance of BankAccount
    Make a deposit using the Deposit() FUNCTION
    Make a withdrawal
    Make a Transfer

END
